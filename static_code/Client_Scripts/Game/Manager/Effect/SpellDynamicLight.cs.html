    public static void RegisterConfig(SpellLightConfig config, params string[] keywords)
    {
        if (config == null) return;
        if (config.SkillId > 0)
        {
            _configsBySkillId[config.SkillId] = config;
        }
        if (keywords != null)
        {
            foreach (string kw in keywords)
            {
                if (!string.IsNullOrEmpty(kw))
                {
                    _configsByKeyword[kw.ToLowerInvariant()] = config;
                }
            }
        }
    }

    public static SpellLightConfig GetConfig(int skillId, string effectClass = null, string skillName = null)
    {
        if (skillId > 0 && _configsBySkillId.TryGetValue(skillId, out SpellLightConfig cfg))
        {
            return cfg;
        }

        if (!string.IsNullOrEmpty(effectClass))
        {
            string lower = effectClass.ToLowerInvariant();
            foreach (KeyValuePair<string, SpellLightConfig> kvp in _configsByKeyword)
            {
                if (lower.Contains(kvp.Key))
                {
                    return kvp.Value;
                }
            }
        }

        if (!string.IsNullOrEmpty(skillName))
        {
            string lowerName = skillName.ToLowerInvariant();
            foreach (KeyValuePair<string, SpellLightConfig> kvp in _configsByKeyword)
            {
                if (lowerName.Contains(kvp.Key))
                {
                    return kvp.Value;
                }
            }
        }

        return null;
    }

    public static SpellLightConfig GetConfig(Skill skill, string effectClass = null)
    {
        if (skill == null) return GetConfig(0, effectClass, null);

        string name = null;
        if (skill.SkillNameDatas != null && skill.SkillNameDatas.Length > 0 && skill.SkillNameDatas[0] != null)
        {
            name = skill.SkillNameDatas[0].Name;
        }

        return GetConfig(skill.SkillId, effectClass, name);
    }

    #region Cast Light Component on Caster

    private Light _castLight;
    private GameObject _castLightObject;
    private SpellLightConfig _currentCastConfig;
    private float _castDuration;
    private float _castStartTime;
    private bool _isCasting;
    private bool _isFadingOut;
    private float _fadeStartTime;
    private float _fadeStartIntensity;
    private const float FadeDuration = 0.2f;

    public void BeginCast(SpellLightConfig config, float durationSec)
    {
        if (config == null) return;

        _currentCastConfig = config;
        _castDuration = Mathf.Max(0.1f, durationSec);
        _castStartTime = Time.time;
        _isCasting = true;
        _isFadingOut = false;

        EnsureCastLightCreated(config.CastHeightOffset);
        if (_castLight != null)
        {
            _castLight.transform.localPosition = new Vector3(0, config.CastHeightOffset, 0);
            _castLight.color = config.CastLightColor;
            _castLight.range = config.CastLightRange;
            _castLight.intensity = 0f;
            _castLight.enabled = true;
        }
    }

    public void EndCast()
    {
        if (!_isCasting && !_isFadingOut) return;

        _isCasting = false;
        _isFadingOut = true;
        _fadeStartTime = Time.time;
        _fadeStartIntensity = _castLight != null ? _castLight.intensity : 0f;
    }

    private void EnsureCastLightCreated(float heightOffset = 0.9f)
    {
        if (_castLightObject == null)
        {
            Transform existing = transform.Find("SpellCastLight");
            if (existing != null)
            {
                _castLightObject = existing.gameObject;
                _castLight = _castLightObject.GetComponent<Light>();
                _castLightObject.transform.localPosition = new Vector3(0, heightOffset, 0);
            }
            else
            {
                _castLightObject = new GameObject("SpellCastLight");
                _castLightObject.transform.SetParent(transform, false);
                _castLightObject.transform.localPosition = new Vector3(0, heightOffset, 0);

                _castLight = _castLightObject.AddComponent<Light>();
                _castLight.type = LightType.Point;
                _castLight.shadows = LightShadows.None;
                _castLight.renderMode = LightRenderMode.Auto;
            }
        }
        else
        {
            _castLightObject.transform.localPosition = new Vector3(0, heightOffset, 0);
        }
    }

    private void Update()
    {
        if (_castLight == null || !_castLight.enabled) return;

        if (_isCasting)
        {
            float elapsed = Time.time - _castStartTime;
            float progress = Mathf.Clamp01(elapsed / _castDuration);

            // Smooth flare-up curve
            float baseIntensity = Mathf.Lerp(0f, _currentCastConfig.CastMaxIntensity, progress);

            // Gentle living pulsation
            float pulse = 1f + 0.12f * Mathf.Sin(Time.time * 16f) + 0.05f * Mathf.Cos(Time.time * 25f);
            _castLight.intensity = baseIntensity * pulse;
        }
        else if (_isFadingOut)
        {
            float elapsed = Time.time - _fadeStartTime;
            float progress = Mathf.Clamp01(elapsed / FadeDuration);

            _castLight.intensity = Mathf.Lerp(_fadeStartIntensity, 0f, progress);

            if (progress >= 1f)
            {
                _isFadingOut = false;
                _castLight.intensity = 0f;
                _castLight.enabled = false;
            }
        }
    }

    #endregion

    #region Static Public API

    /// <summary>
    /// Starts dynamic casting light on the caster entity.
    /// </summary>
    public static void StartCastLight(Entity caster, Skill skill, int hitTimeMs)
    {
        if (caster == null || skill == null) return;

        string effectClass = (skill.SkillEffect != null && skill.SkillEffect.CastingActions != null && skill.SkillEffect.CastingActions.Count > 0)
            ? skill.SkillEffect.CastingActions[0].EffectClass
            : null;

        SpellLightConfig config = GetConfig(skill, effectClass);
        if (config == null) return;

        SpellDynamicLight comp = caster.GetComponent<SpellDynamicLight>();
        if (comp == null)
        {
            comp = caster.gameObject.AddComponent<SpellDynamicLight>();
        }

        comp.BeginCast(config, hitTimeMs / 1000f);
    }

    /// <summary>
    /// Smoothly fades out and disables casting light on the caster entity.
    /// </summary>
    public static void StopCastLight(Entity caster)
    {
        if (caster == null) return;

        SpellDynamicLight comp = caster.GetComponent<SpellDynamicLight>();
        if (comp != null)
        {
            comp.EndCast();
        }
    }

    /// <summary>
    /// Attaches or enables a dynamic point light on a flying spell projectile.
    /// </summary>
    public static void AttachProjectileLight(PooledEffect effect)
    {
        if (effect == null || effect.GameObject == null) return;

        SpellLightConfig config = GetConfig(effect.Skill, effect.EffectClass);
        if (config == null) return;

        Transform lightTransform = effect.GameObject.transform.Find("SpellProjectileLight");
        Light lightComp;

        if (lightTransform == null)
        {
            GameObject lightGo = new GameObject("SpellProjectileLight");
            lightGo.transform.SetParent(effect.GameObject.transform, false);
            lightGo.transform.localPosition = Vector3.zero;

            lightComp = lightGo.AddComponent<Light>();
            lightComp.type = LightType.Point;
            lightComp.shadows = LightShadows.None;
            lightComp.renderMode = LightRenderMode.Auto;
        }
        else
        {
            lightComp = lightTransform.GetComponent<Light>();
        }

        if (lightComp != null)
        {
            lightComp.color = config.ProjectileLightColor;
            lightComp.range = config.ProjectileLightRange;
            lightComp.intensity = config.ProjectileIntensity;
            lightComp.shadows = LightShadows.None;
            lightComp.enabled = true;
            lightComp.gameObject.SetActive(true);
        }
    }

    /// <summary>
    /// Triggers an explosive impact flash at the target position.
    /// </summary>
    public static void TriggerHitFlash(Entity target, Skill skill, string effectClass, Vector3 fallbackPosition)
    {
        SpellLightConfig config = GetConfig(skill, effectClass);
        if (config == null) return;

        Vector3 hitPos = fallbackPosition;
        if (target != null)
        {
            float heightOffset = (target.Appearance != null && target.Appearance.CollisionHeight > 0.1f)
                ? target.Appearance.CollisionHeight * 0.5f
                : 1.0f;
            hitPos = target.transform.position + Vector3.up * heightOffset;
        }

        Instance.SpawnFlashLight(config, hitPos);
    }

    /// <summary>
    /// Triggers an instant flash (e.g. for heals, buffs, or self-skills with no flying projectile).
    /// </summary>
    public static void TriggerSelfFlash(Entity target, Skill skill)
    {
        if (target == null || skill == null) return;
        TriggerHitFlash(target, skill, null, target.transform.position);
    }

    #endregion

    #region Hit Flash Pooling

    private readonly Queue<Light> _flashLightPool = new Queue<Light>();
    private Transform _flashLightContainer;

    private void SpawnFlashLight(SpellLightConfig config, Vector3 position)
    {
        if (_flashLightContainer == null)
        {
            GameObject container = new GameObject("HitFlashPool");
            container.transform.SetParent(transform, false);
            _flashLightContainer = container.transform;
        }

        Light flashLight = null;
        while (_flashLightPool.Count > 0)
        {
            Light candidate = _flashLightPool.Dequeue();
            if (candidate != null)
            {
                flashLight = candidate;
                break;
            }
        }

        if (flashLight == null)
        {
            GameObject go = new GameObject("SpellHitFlash");
            go.transform.SetParent(_flashLightContainer, false);
            flashLight = go.AddComponent<Light>();
            flashLight.type = LightType.Point;
            flashLight.shadows = LightShadows.None;
            flashLight.renderMode = LightRenderMode.Auto;
        }

        flashLight.transform.position = position;
        flashLight.color = config.HitFlashColor;
        flashLight.range = config.HitFlashRange;
        flashLight.intensity = config.HitFlashPeakIntensity;
        flashLight.enabled = true;
        flashLight.gameObject.SetActive(true);

        StartCoroutine(AnimateFlashRoutine(flashLight, config));
    }

    private IEnumerator AnimateFlashRoutine(Light flashLight, SpellLightConfig config)
    {
        float peakIntensity = config.HitFlashPeakIntensity;
        float holdTime = config.HitFlashHoldTime;
        float fadeTime = config.HitFlashFadeTime;

        // 1. Hold peak intensity
        float elapsed = 0f;
        while (elapsed < holdTime)
        {
            elapsed += Time.deltaTime;
            yield return null;
        }

        // 2. Smooth fade out
        elapsed = 0f;
        while (elapsed < fadeTime)
        {
            elapsed += Time.deltaTime;
            float t = Mathf.Clamp01(elapsed / fadeTime);
            if (flashLight != null)
            {
                flashLight.intensity = Mathf.Lerp(peakIntensity, 0f, t);
            }
            yield return null;
        }

        // 3. Return to pool
        if (flashLight != null)
        {
            flashLight.intensity = 0f;
            flashLight.enabled = false;
            flashLight.gameObject.SetActive(false);
            _flashLightPool.Enqueue(flashLight);
        }
    }

    #endregion
